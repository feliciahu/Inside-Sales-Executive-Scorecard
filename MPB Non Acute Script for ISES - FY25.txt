CREATE VIEW SBX_PSAS_DB.SALES_OPS_GOV.V_ISES_ISAM_MPB_NON_ACUTE AS 
SELECT 'CPH' AS SEGMENT,
'CPH MPB' AS TEAM,
'' AS TERR_GRP,
CASE WHEN YR_MTH = '2024-04-01' THEN '2024-04'
     WHEN YR_MTH = '2024-05-01' THEN '2024-05'
     WHEN YR_MTH = '2024-06-01' THEN '2024-06'
     WHEN YR_MTH = '2024-07-01' THEN '2024-07'
     WHEN YR_MTH = '2024-08-01' THEN '2024-08'
     WHEN YR_MTH = '2024-09-01' THEN '2024-09'
     WHEN YR_MTH = '2024-10-01' THEN '2024-10'
     WHEN YR_MTH = '2024-11-01' THEN '2024-11'
     WHEN YR_MTH = '2024-12-01' THEN '2024-12'
     WHEN YR_MTH = '2025-01-01' THEN '2025-01'
     WHEN YR_MTH = '2025-02-01' THEN '2025-02'
     ELSE '2025-03' END AS YR_MONTH,
COUNT(DISTINCT CUST_ACCT_ID) AS CUST_CNT,
SUM(TARGET_GROSS_PROFIT) AS GROSS_PROFIT_GOAL,
SUM(TARGET_NET_SALES) AS NET_SLS_GOAL,
SUM(GROSS_PROFIT) AS GROSS_PROFIT_ACTUAL, 
SUM(NET_SLS) AS NET_SLS_ACTUAL
FROM SBX_PSAS_DB.SALES_OPS_GOV.MPB_NON_ACUTE_FY25_ACTUALS
GROUP BY 1,2,3,4;
